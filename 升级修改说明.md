```js
function parsePlanSignal(text) {
  const m = text.match(/<PLAN_SIGNAL>([\s\S]*?)<\/PLAN_SIGNAL>/);
  if (!m) return null;
  try { return JSON.parse(m[1].trim()); } catch { return null; }
}

function stripSignal(text) {
  return text.replace(/<PLAN_SIGNAL>[\s\S]*?<\/PLAN_SIGNAL>/g, "").trim();
}
```

```js
async function searchGaode(tags, apiKey) {
  const prefMap = {
    "户外": "公园 景区", "亲子": "亲子乐园", "文化": "博物馆 历史景点",
    "美食": "特色餐厅", "运动": "运动公园", "室内": "购物中心 展馆"
  };
  const kw = (tags.preferences || ["景点"]).map(p => prefMap[p] || p).join(" ");
  const base = `https://restapi.amap.com/v3/place/text?city=${encodeURIComponent(tags.city)}&key=${apiKey}&output=JSON&offset=6&extensions=base`;
  const [r1, r2] = await Promise.all([
    fetch(`${base}&keywords=${encodeURIComponent(kw)}`),
    fetch(`${base}&keywords=${encodeURIComponent("餐厅 美食")}`)
  ]);
  const [d1, d2] = await Promise.all([r1.json(), r2.json()]);
  return {
    activityPlaces: (d1.pois || []).map(p => ({ name: p.name, address: p.address, type: p.type })),
    restaurants: (d2.pois || []).map(p => ({ name: p.name, address: p.address, type: p.type }))
  };
}
```

```js
const startPlanningFlow = async (tags) => {
  // 阶段1：分析需求
  setPhase("extracting");
  setStatusMsg("正在分析你的需求...");
  await new Promise(r => setTimeout(r, 1000));

  // 阶段2：搜索地点
  setPhase("searching");
  setStatusMsg(`正在搜索${tags.city || "附近"}的真实地点...`);
  
  let placeData;
  if (gaodeKey.trim()) {
    try {
      placeData = await searchGaode(tags, gaodeKey.trim());
      setUseRealData(true);
    } catch (e) {
      placeData = await generatePlacesClaude(tags); // 降级
    }
  } else {
    placeData = await generatePlacesClaude(tags);
  }
  await new Promise(r => setTimeout(r, 400));

  // 阶段3：生成方案
  setPhase("planning");
  setStatusMsg("正在生成并评分活动方案...");
  
  const planPrompt = `用户需求：${JSON.stringify(tags)}

地点数据：
活动地点：${JSON.stringify(placeData.activityPlaces)}
餐饮选择：${JSON.stringify(placeData.restaurants)}

请生成2-3个活动方案并评分对比。`;

  const planText = await callClaude(
    [{ role: "user", content: planPrompt }],
    "PLAN_SYSTEM",  // 使用 PLAN_SYSTEM_PROMPT
    2000
  );

  try {
    const parsed = JSON.parse(planText.replace(/```json|```/g, "").trim());
    setPlans(parsed.plans || []);
    setRecommendedIdx(parsed.recommended_index || 0);
    setRecommendReason(parsed.recommendation_reason || "");
    setExpandedPlan(parsed.recommended_index || 0);
  } catch {
    setPlans([]);
  }
  
  setPhase("result");
  setStatusMsg("");
};
```

```jsx
        width: "100%", border: "none", borderRadius: 8, padding: "8px", fontSize: 13, fontWeight: 600, cursor: "pointer",
        background: expanded ? "#F5F5F5" : "#1A1A1A",
        color: expanded ? "#666" : "#FFBE00"
      }}>
        {expanded ? "收起行程 ▲" : "查看详细行程 ▼"}
      </button>
      {/* 行程时间线 */}
      {expanded && (
        <div style={{ marginTop: 12 }}>
          {(plan.itinerary || []).map((item, i) => (
            <div key={i} style={{ display: "flex", gap: 10, marginBottom: 12 }}>
              <div style={{ display: "flex", flexDirection: "column", alignItems: "center" }}>
                <div style={{ width: 34, height: 34, borderRadius: "50%", background: "#FFBE00", display: "flex", alignItems: "center", justifyContent: "center", fontSize: 11, fontWeight: 800, color: "#1A1A1A", flexShrink: 0 }}>
                  {item.time}
                </div>
                {i < plan.itinerary.length - 1 && (
                  <div style={{ width: 2, flex: 1, background: "#FFE566", margin: "3px 0", minHeight: 12 }} />
                )}
              </div>
              <div style={{ flex: 1 }}>
                <div style={{ fontWeight: 700, fontSize: 14, color: "#1A1A1A" }}>{item.place}</div>
                <div style={{ fontSize: 11, color: "#AAA", margin: "2px 0" }}>⏱ {item.duration}</div>
                <div style={{ fontSize: 13, color: "#444", lineHeight: 1.65 }}>{item.desc}</div>
                {item.tips && <div style={{ fontSize: 12, color: "#C8960A", marginTop: 4 }}>💡 {item.tips}</div>}
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  </div>
);
```

### 模块8：修改 JSX 渲染部分

在原有 return 语句中，按以下逻辑修改：

**Header 区域**：在右侧加一个「⚙️ 高德Key」按钮，点击展开 input 框供用户填入高德 API Key；若 `useRealData` 为 true，显示「📡 真实数据」徽标。

**Header 下方**：插入 `<PhaseBar phase={phase} />`

**主内容区**：根据 phase 判断显示内容：
- `phase === "chat"` → 显示原有聊天界面
- `phase === "extracting" / "searching" / "planning"` → 显示居中 loading 动画 + statusMsg 文字
- `phase === "result"` → 显示：`<TagChips />` + 推荐理由文字块 + `plans.map(<PlanCard />)` + 「🔄 重新规划」按钮

**重新规划按钮** 点击后重置所有 state 回初始值。

## 三、整体流程说明（给 ChatGPT 理解用）

```
用户开始对话
    ↓
Claude（CHAT_SYSTEM）逐步收集：城市、时长、人群、偏好
    ↓
收集充分后，Claude 在回复末尾输出 <PLAN_SIGNAL>{...}</PLAN_SIGNAL>
    ↓
前端检测到信号 → 解析 JSON 标签 → 切换 phase
    ↓
phase: extracting → 展示「分析中」动画（1秒）
    ↓
phase: searching → 调用高德 API（有 Key）或 Claude 生成参考地点（无 Key）
    ↓
phase: planning → 把需求 + 地点数据发给 Claude（PLAN_SYSTEM），要求输出 JSON 方案
    ↓
phase: result → 渲染 TagChips + 2-3个 PlanCard（含评分条 + 可展开行程）
```
