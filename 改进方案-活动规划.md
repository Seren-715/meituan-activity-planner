# 活动规划改进方案

## 一、当前项目分析

### 现有优势
- ✅ 基础架构完整（FastAPI + React）
- ✅ 支持家庭/朋友场景
- ✅ 有主方案+备选方案
- ✅ 有路线摘要和执行动作

### 需要改进的地方
1. **时间管理粗糙** — 没有缓冲时间，时间衔接不够精细
2. **缺少费用估算** — 用户不知道要花多少钱
3. **备选方案单一** — 只有活动+餐厅组合，缺少场景化备选
4. **执行动作简单** — 缺少预约时间、等待提醒等细节
5. **用户体验不足** — 缺少可视化时间轴、费用明细等

#### 2.2 费用估算
```python
def _estimate_price(self, candidate: Candidate, group_size: int) -> float:
    """估算单个活动的费用"""
    base_price = candidate.price_per_person * group_size

    # 根据价格等级调整
    price_multipliers = {
        "免费": 0,
        "低价": 0.8,
        "中等": 1.0,
        "高价": 1.3,
        "豪华": 1.8,
    }
    multiplier = price_multipliers.get(candidate.price_level, 1.0)

    return round(base_price * multiplier, 2)

def _build_price_breakdown(self, goal, activity, restaurant, addon) -> list[str]:
    """生成费用明细"""
    breakdown = []
    activity_price = self._estimate_price(activity, goal.group_size)
    breakdown.append(f"{activity.name}: ¥{activity_price:.0f}")

    restaurant_price = self._estimate_price(restaurant, goal.group_size)
    breakdown.append(f"{restaurant.name}: ¥{restaurant_price:.0f}")

    if addon:
        addon_price = self._estimate_price(addon, goal.group_size)
        breakdown.append(f"{addon.name}: ¥{addon_price:.0f}")

    return breakdown
```

#### 2.3 天气感知规划
```python
def _check_weather_dependency(self, candidate: Candidate, weather_is_bad: bool) -> str:
    """检查天气对活动的影响"""
    if not candidate.weather_dependent:
        return ""

    if weather_is_bad and not candidate.indoor:
        return f"⚠️ {candidate.name}为户外活动，下雨天建议更换为室内替代"

    return ""

def _build_weather_alerts(self, goal, activity, restaurant, addon) -> list[str]:
    """生成天气相关提醒"""
    alerts = []
    if goal._weather_is_bad:
        for candidate in [activity, restaurant, addon]:
            if candidate and candidate.weather_dependent and not candidate.indoor:
                alerts.append(f"🌧️ {candidate.name}受天气影响，建议准备雨具或更换方案")
    return alerts
```

#### 2.4 拥挤度优化
```python
def _adjust_for_crowd(self, candidate: Candidate, time_window: str) -> float:
    """根据拥挤程度调整评分"""
    score_adjustment = 0.0

    if candidate.crowd_level == "high":
        # 高峰时段减分
        if time_window in ["下午", "晚上"]:
            score_adjustment -= 5.0
        # 建议错峰
        candidate.reason += "（建议错峰前往）"

    elif candidate.crowd_level == "low":
        score_adjustment += 3.0

    return score_adjustment
```

### 改进 3：场景化备选方案

#### 3.1 天气备选
```python
def _build_weather_fallback(self, primary_activity, weather_is_bad):
    """构建天气备选方案"""
    if not weather_is_bad:
        return None

    # 如果主活动是户外，找室内替代
    if primary_activity.weather_dependent and not primary_activity.indoor:
        indoor_alternatives = [
            c for c in self.all_candidates
            if c.category == "activity"
            and c.indoor
            and c.family_friendly == primary_activity.family_friendly
        ]
        if indoor_alternatives:
            return indoor_alternatives[0]

    return None
```

#### 3.2 时间备选

def _build_time_fallback(self, goal, total_minutes):
    """如果时间超支，建议跳过或压缩某环节"""
    target_minutes = goal.duration_hours * 60

    if total_minutes > target_minutes + 30:
        return [
            "⏰ 时间略超预期，建议：",
            "1. 压缩用餐时间（改为快餐或外卖）",
            "2. 跳过补充活动，直接用餐",
            "3. 选择更近的餐厅减少路程",
        ]

    return []
```

### 改进 4：增强执行动作

#### 4.1 执行动作增加细节

@dataclass(slots=True)
class ExecutionAction:
    action_type: ActionType
    target: str
    payload: dict[str, str]

    # 新增字段 ⭐
    timing: str = "now"                   # 执行时机（now/before_2h/before_1day）
    required: bool = True                 # 是否必须执行
    fallback: str = ""                    # 失败后的备选
    estimated_wait: str = ""              # 预计等待时间
    user_confirmation: bool = False       # 是否需要用户确认
    cost_estimate: float = 0.0            # 预估费用
```

#### 4.2 智能执行编排
```python
def _build_actions(self, goal, itinerary):
    """构建更智能的执行动作"""
    actions = []

    # 活动预约（如果需要）
    activity_stop = next(s for s in itinerary.stops if s.category == "activity")
    if activity_stop.booking_required:
        actions.append(ExecutionAction(
            action_type="reserve",
            target=activity_stop.title,
            timing="before_2h",  # 提前2小时预约
            required=True,
            fallback="现场排队",
            estimated_wait="约15分钟",
            user_confirmation=True,
            payload={
                "time": activity_stop.start_time,
                "group_size": str(goal.group_size),
                "contact": "需要提供联系方式",
            }
        ))

    # 餐厅排号
    restaurant_stop = next(s for s in itinerary.stops if s.category == "restaurant")
    actions.append(ExecutionAction(
        action_type="queue",
        target=restaurant_stop.title,
        timing="before_30min",  # 提前30分钟排号
        required=True,
        fallback="换一家餐厅",
        estimated_wait="约20分钟",
        payload={
            "group_size": str(goal.group_size),
            "scene": goal.scene,
        }
    ))

    # 家庭场景：配送蛋糕
    if goal.scene == "family":
        actions.append(ExecutionAction(
            action_type="delivery",
            target="餐后甜点配送",
            timing="during_meal",  # 用餐时下单
            required=False,
            fallback="餐厅内甜点",
            payload={
                "restaurant": restaurant_stop.title,
                "delivery_time": "用餐结束前30分钟",
            }
        ))

    # 朋友场景：预点饮品
    else:
        actions.append(ExecutionAction(
            action_type="order",
            target="聚会饮品预点",
            timing="before_1h",  # 提前1小时
            required=False,
            fallback="现场点单",
            payload={
                "restaurant": restaurant_stop.title,
            }
        ))

    # 分享
    actions.append(ExecutionAction(
        action_type="share",
        target=goal.share_target,
        timing="after_plan",  # 规划完成后
        required=False,
        payload={
            "scene": goal.scene,
            "score": f"{itinerary.score:.1f}",
        }
    ))

    return actions
```
